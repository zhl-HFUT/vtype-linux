cmake_minimum_required(VERSION 3.16)
project(fcitx5-wetype VERSION 1.0 LANGUAGES CXX)

include(GNUInstallDirs)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Fcitx5Core REQUIRED)

# 显式指定的包内启动器可能在安装阶段才生成，配置时不能丢弃该路径。
find_program(WETYPE_PYTHON_EXECUTABLE NAMES python3)
if(NOT WETYPE_PYTHON_EXECUTABLE)
    message(FATAL_ERROR "找不到 python3；可用 -DWETYPE_PYTHON_EXECUTABLE=... 指定")
endif()

add_library(fcitx5-wetype MODULE src/wetype.cpp)
target_compile_definitions(fcitx5-wetype PRIVATE WETYPE_PYTHON_EXECUTABLE="${WETYPE_PYTHON_EXECUTABLE}")
target_include_directories(fcitx5-wetype PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../harness ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(fcitx5-wetype PRIVATE Fcitx5::Core)

include(CTest)
if(BUILD_TESTING)
    add_executable(double-pinyin-test tests/double_pinyin_test.cpp)
    target_include_directories(double-pinyin-test PRIVATE ../harness)
    add_test(NAME double_pinyin COMMAND double-pinyin-test)
    add_executable(glossary-test tests/glossary_test.cpp)
    target_include_directories(glossary-test PRIVATE src)
    add_test(NAME glossary COMMAND glossary-test)
    add_executable(punctuation-test tests/punctuation_test.cpp)
    target_include_directories(punctuation-test PRIVATE src)
    target_link_libraries(punctuation-test PRIVATE Fcitx5::Core)
    add_test(NAME punctuation COMMAND punctuation-test)
    add_executable(harness-log-test tests/harness_log_test.cpp)
    target_include_directories(harness-log-test PRIVATE src)
    add_test(NAME harness_log COMMAND harness-log-test)
    add_executable(voice-protocol-test tests/voice_protocol_test.cpp)
    target_include_directories(voice-protocol-test PRIVATE src)
    add_test(NAME voice_protocol COMMAND voice-protocol-test)
    add_executable(common-phrases-test tests/common_phrases_test.cpp)
    target_include_directories(common-phrases-test PRIVATE src)
    add_test(NAME common_phrases COMMAND common-phrases-test)
    add_executable(paging-config-test tests/paging_config_test.cpp)
    target_include_directories(paging-config-test PRIVATE src ../harness)
    target_link_libraries(paging-config-test PRIVATE Fcitx5::Core)
    add_test(NAME paging_config COMMAND paging-config-test)
    add_executable(fuzzy-pinyin-test tests/fuzzy_pinyin_test.cpp)
    target_include_directories(fuzzy-pinyin-test PRIVATE src ../harness)
    target_link_libraries(fuzzy-pinyin-test PRIVATE Fcitx5::Core)
    add_test(NAME fuzzy_pinyin COMMAND fuzzy-pinyin-test)
    add_executable(cursor-test tests/cursor_test.cpp)
    target_include_directories(cursor-test PRIVATE src ../harness)
    target_link_libraries(cursor-test PRIVATE Fcitx5::Core)
    add_test(NAME cursor COMMAND cursor-test)
endif()

install(TARGETS fcitx5-wetype LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
install(FILES wetype-addon.conf RENAME wetype.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon)
install(FILES wetype-im.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod)
